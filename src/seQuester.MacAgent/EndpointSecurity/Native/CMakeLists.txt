cmake_minimum_required(VERSION 3.24)
project(sequester_es LANGUAGES CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "Requires macOS and the Apple EndpointSecurity SDK")
endif()
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0")
add_library(sequester_es SHARED sq_es.mm)
target_compile_features(sequester_es PRIVATE cxx_std_17)
target_compile_options(sequester_es PRIVATE -fobjc-arc -fblocks -Wall -Wextra -Werror)
find_library(ENDPOINT_SECURITY_LIBRARY EndpointSecurity REQUIRED)
target_link_libraries(sequester_es PRIVATE ${ENDPOINT_SECURITY_LIBRARY} "-framework Foundation" bsm)
set_target_properties(sequester_es PROPERTIES CXX_VISIBILITY_PRESET default)
include(CTest)
if(BUILD_TESTING)
  add_executable(sq_es_native_tests tests.mm)
  target_compile_features(sq_es_native_tests PRIVATE cxx_std_17)
  target_compile_options(sq_es_native_tests PRIVATE -fobjc-arc -fblocks -Wall -Wextra -Werror)
  target_link_libraries(sq_es_native_tests PRIVATE ${ENDPOINT_SECURITY_LIBRARY} "-framework Foundation" bsm)
  add_test(NAME sdk_snapshot_without_client COMMAND sq_es_native_tests)
endif()
